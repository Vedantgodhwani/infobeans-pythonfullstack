num_str = input("Enter Number: ")
alternating_sum = 0

for i in range(len(num_str)):
    digit = int(num_str[i])
    
    if i == 0:
        alternating_sum = digit
    elif i % 2 == 1:
        # Subtract for indices 1, 3, 5...
        alternating_sum -= digit
    else:
        alternating_sum += digit

print(f"Result = {alternating_sum}")

if alternating_sum > 0:
    print("Positive Pattern")
elif alternating_sum < 0:
    print("Negative Pattern")
else:
    print("Zero/Neutral Pattern")
